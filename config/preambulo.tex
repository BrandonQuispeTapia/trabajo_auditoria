% Microtype: mejora sutilmente el espaciado.

% requiere tablas específicas que la familia newtxtext no declara, lo que

% scalable...»). El protrusion (ajuste de márgenes en signos de puntuación),
% que sí es compatible, se mantiene activo.

% opciones distintas («table» y luego «dvipsnames»), lo que produce un
% error de «Option clash for package xcolor». Se unifica en una sola
% carga con ambas opciones.

\usepackage{pdflscape}
\usepackage{rotating}

% ==========================================
% 3. IDIOMA Y TABLAS
% ==========================================
\usepackage[spanish,es-nodecimaldot,es-lcroman]{babel}

}

% ==========================================
% 5. GRÁFICOS Y DIAGRAMAS (TIKZ)
% ==========================================
% AJUSTE: la plantilla original cargaba tikz y sus librerías tres veces
% en bloques distintos (con paquetes de opciones ligeramente diferentes).
% Se consolida en una sola carga con todas las librerías necesarias:
%   - shapes.geometric, arrows, positioning, shadows, arrows.meta: las
%     que pedía la plantilla original.
%   - babel: IMPRESCINDIBLE. babel-spanish activa los caracteres < y >

\tikzstyle{line} = [draw, -latex', thick]

% ==========================================
% AJUSTE — ADAPTADOR DE JERARQUÍA DE TÍTULOS
%
% Los archivos de secciones/ se escriben con \section para el título de
% capítulo, \subsection para las secciones y \subsubsection para las
% subsecciones (así se redactaron los diez capítulos del informe). Esta
% clase (report) ya tiene un \chapter nativo, así que aquí NO se crea
% ningún comando nuevo: se redirige un nivel cada uno de los que ya usan
% los archivos hacia el nivel real que le corresponde. Los formatos
% definidos arriba con titlesec (para \chapter y \section reales) se
% aplican automáticamente a través de este adaptador.
% ==========================================
\makeatletter
\let\ArchivoSection\section
\let\ArchivoSubsection\subsection
\let\ArchivoSubsubsection\subsubsection

\renewcommand{\section}{\chapter}
\renewcommand{\subsection}{\ArchivoSection}
\renewcommand{\subsubsection}{\ArchivoSubsection}
\makeatother

% Profundidad de numeración: capítulo.sección.subsección (3 niveles).
\setcounter{secnumdepth}{2}
% Profundidad del índice: solo hasta «sección» (no baja a subsección).
\setcounter{tocdepth}{1}
